\documentclass[tikz,border=6mm]{standalone}
\usepackage{lua-tikz3dtools}

\def\VIEWA{1.10*pi}
\def\VIEWB{.36*pi}

\begin{document}
\foreach \frame in {5,...,33} {
    \pgfmathsetmacro{\ANIMT}{\frame/39}
    \begin{tikzpicture}
        \useasboundingbox (-3.15,-2.35) rectangle (3.15,2.45);
        \ltdtsetobject[
            name={return "scene"},
            object={
                local t=\ANIMT
                local xmin,xmax=1,3
                local ymin,ymax=.5,2.5
                local zmin,zmax=.5,2.5
                local x0=xmin+t*(xmax-xmin)
                local y0=ymin+t*(ymax-ymin)
                local scale=1.58
                local center=Vector:new{2,1.5,1.5,1}
                local function f(x,y)
                    return -(x-2)^3-(y-1.5)^2+2
                end
                local function localize(p)
                    return p:hsub(center):hscale(scale)
                end
                local function unlocalize(p)
                    return p:hscale(1/scale):hadd(center)
                end
                local function graph(x,y)
                    return localize(Vector:new{x,y,f(x,y),1})
                end
                local slope=-2*(y0-1.5)
                local function tangentz(y)
                    return f(x0,y0)+slope*(y-y0)
                end
                local function inside(p)
                    local q=unlocalize(p)
                    local e=1e-7
                    return q[1]>=xmin-e and q[1]<=xmax+e
                        and q[2]>=ymin-e and q[2]<=ymax+e
                        and q[3]>=zmin-e and q[3]<=zmax+e
                end
                local function edge(a,b)
                    return {
                        localize(Vector:new{a[1],a[2],a[3],1}),
                        localize(Vector:new{b[1],b[2],b[3],1})
                    }
                end
                local edges={
                    edge({xmin,ymin,zmin},{xmax,ymin,zmin}),
                    edge({xmax,ymin,zmin},{xmax,ymax,zmin}),
                    edge({xmax,ymax,zmin},{xmin,ymax,zmin}),
                    edge({xmin,ymax,zmin},{xmin,ymin,zmin}),
                    edge({xmin,ymin,zmax},{xmax,ymin,zmax}),
                    edge({xmax,ymin,zmax},{xmax,ymax,zmax}),
                    edge({xmax,ymax,zmax},{xmin,ymax,zmax}),
                    edge({xmin,ymax,zmax},{xmin,ymin,zmax}),
                    edge({xmin,ymin,zmin},{xmin,ymin,zmax}),
                    edge({xmax,ymin,zmin},{xmax,ymin,zmax}),
                    edge({xmax,ymax,zmin},{xmax,ymax,zmax}),
                    edge({xmin,ymax,zmin},{xmin,ymax,zmax})
                }
                local view=Matrix.zyzrotation(Vector:new{\VIEWA,\VIEWB,pi/2})
                    :multiply(Matrix.perspective(Vector:new{0,0,-1/18,1}))
                local p=localize(Vector:new{x0,y0,f(x0,y0),1})
                return {
                    xmin=xmin,
                    xmax=xmax,
                    ymin=ymin,
                    ymax=ymax,
                    zmin=zmin,
                    zmax=zmax,
                    x0=x0,
                    y0=y0,
                    f=f,
                    tangentz=tangentz,
                    localize=localize,
                    graph=graph,
                    inside=inside,
                    edges=edges,
                    view=view,
                    inv=view:inverse(),
                    p=p
                }
            }
        ]
        \ltdtappendsolid[
            uparams={return Vector:new{scene.xmin,scene.xmax,2}},
            vparams={return Vector:new{scene.ymin,scene.ymax,2}},
            wparams={return Vector:new{scene.zmin,scene.zmax,2}},
            v={return scene.localize(Vector:new{u,v,w,1})},
            transformation={return scene.view},
            filter={return false}
        ]
        \ltdtappendsurface[
            name=slice,
            uparams={return Vector:new{scene.zmin-.4,scene.zmax+.4,2}},
            vparams={return Vector:new{scene.ymin-.4,scene.ymax+.4,2}},
            v={return scene.localize(Vector:new{scene.x0,v,u,1})},
            transformation={return scene.view},
            fill options={return "fill=cyan!7!white,fill opacity=.40,draw=none"},
            filter={
                local p=A:hadd(B):hadd(C):hscale(1/3):multiply(scene.inv)
                return scene.inside(p)
            },
            curve={
                local F="local p=A:hadd(B):hscale(1/2):multiply(scene.inv) return scene.inside(p)"
                local G="draw=blue!32!gray,line width=.11pt,line cap=round"
                return {
                    {
                        start=Vector:new{scene.zmin-.4,1},
                        stop=Vector:new{scene.zmax+.4,1},
                        drawoptions=G,
                        filter=F
                    },
                    {
                        start=Vector:new{scene.zmin-.4,1.5},
                        stop=Vector:new{scene.zmax+.4,1.5},
                        drawoptions=G,
                        filter=F
                    },
                    {
                        start=Vector:new{scene.zmin-.4,2},
                        stop=Vector:new{scene.zmax+.4,2},
                        drawoptions=G,
                        filter=F
                    },
                    {
                        start=Vector:new{1,scene.ymin-.4},
                        stop=Vector:new{1,scene.ymax+.4},
                        drawoptions=G,
                        filter=F
                    },
                    {
                        start=Vector:new{1.5,scene.ymin-.4},
                        stop=Vector:new{1.5,scene.ymax+.4},
                        drawoptions=G,
                        filter=F
                    },
                    {
                        start=Vector:new{2,scene.ymin-.4},
                        stop=Vector:new{2,scene.ymax+.4},
                        drawoptions=G,
                        filter=F
                    },
                    {
                        start=Vector:new{scene.tangentz(scene.ymin-.5),scene.ymin-.5},
                        stop=Vector:new{scene.tangentz(scene.ymax+.5),scene.ymax+.5},
                        drawoptions="draw=black!90,line width=.52pt,line cap=round",
                        filter=F
                    }
                }
            }
        ]
        \ltdtappendsurface[
            uparams={return Vector:new{.65,3.35,7}},
            vparams={return Vector:new{.15,2.85,7}},
            adaptive={return {tolerance=.005,feature_tolerance=.002,max_depth=9,max_triangles=700}},
            v={return scene.graph(u,v)},
            transformation={return scene.view},
            fill options={return "fill=red!70!orange!88!white,draw=red!48!black,line width=.065pt,line join=round"},
            partition by={return {{surface="slice",keep="both"}}},
            filter={
                local p=A:hadd(B):hadd(C):hscale(1/3):multiply(scene.inv)
                return scene.inside(p)
            },
            curve={
                local F="local p=A:hadd(B):hscale(1/2):multiply(scene.inv) return scene.inside(p)"
                return {{
                    start=Vector:new{scene.x0,.15},
                    stop=Vector:new{scene.x0,2.85},
                    drawoptions="draw=green!42!black,line width=.46pt,line cap=round",
                    filter=F
                }}
            }
        ]
        \ltdtappendsurface[
            uparams={return Vector:new{0,tau,10}},
            vparams={return Vector:new{0,pi,6}},
            v={return scene.p:hadd(Vector.sphere(Vector:new{u,v,.05}))},
            transformation={return scene.view},
            fill options={return "fill=yellow!84!orange,draw=orange!60!black,line width=.05pt"}
        ]
        \foreach \i in {1,...,12} {
            \ltdtappendcurve[
                uparams={return Vector:new{0,1,2}},
                v={
                    local e=scene.edges[\i]
                    return e[1]:hscale(1-u):hadd(e[2]:hscale(u))
                },
                transformation={return scene.view},
                draw options={return "draw=black!28,line width=.20pt,line cap=round"}
            ]
        }
        \ltdtdisplaysimplices
    \end{tikzpicture}
}
\end{document}
