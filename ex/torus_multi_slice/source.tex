\documentclass[tikz,border=1cm]{standalone}
\usepackage{lua-tikz3dtools} % https://github.com/Pseudonym321/lua-tikz3dtools

\begin{document}
\foreach \i in {1,...,24} {
\begin{tikzpicture}
    \useasboundingbox[scale=0.5] (-5,-5) rectangle (5,5);
    \ltdtsetobject[
        name={return "objs"},
        object={
            local num=1
            local num2=2
            local function sideS(p)
                return p[2]-num
            end
            local function sideO(p)
                return 3*p[3]-p[2]
            end
            local function curve_visible(p)
                return sideO(p)<0
            end
            return {
                view=Matrix.zyzrotation(
                    Vector:new{pi/6+\i/24*2*pi,pi/3,pi/2}
                ),
                num=num,
                num2=num2,
                sideS=sideS,
                sideO=sideO,
                curve_visible=curve_visible
            }
        }
    ]
    \ltdtappendlight[
        v={return Vector:new{1,1,1,1}}
    ]
    \ltdtappendsurface[
        name=S,
        uparams={return Vector:new{-5,5,10}},
        vparams={return Vector:new{-5,5,10}},
        v={return Vector:new{u,objs.num,v,1}},
        transformation={return objs.view}
    ]
    \ltdtappendsurface[
        name=O,
        uparams={return Vector:new{-5,5,10}},
        vparams={return Vector:new{-5,5,10}},
        v={
            return Vector:new{u,v,v/3,1}:hadd(
                Vector:new{1,1,1/3,1}:hscale(objs.num2)
            )
        },
        transformation={return objs.view}
    ]
    \ltdtappendsurface[
        uparams={return Vector:new{0,1,21}},
        vparams={return Vector:new{0,1,21}},
        v={
            return Vector.sphere(
                Vector:new{u*tau,v*tau,1}
            ):hadd(
                Vector.sphere(
                    Vector:new{u*tau,pi/2,2}
                )
            )
        },
        transformation={return objs.view},
        fill options={
            local p=A:hadd(B):hadd(C):hscale(1/3)
                :multiply(objs.view:inverse())
            local shade=floor(35+45*lighting)
            if objs.sideO(p)>0.01 then
                return "fill=white,fill opacity=0,draw=none,line width=.18pt"
            elseif objs.sideS(p)>0 then
                return "fill=white,fill opacity=0,draw=blue!55!black,line width=.18pt"
            else
                return "fill=white!"..tostring(shade).."!black,draw=black!0,line width=0pt"
            end
        },
        partition by={
            local S_options=
                "draw=red!78!black,line width=.8pt"
            local O_options=
                "draw=green!55!black,line width=.8pt"
            local curve_filter=[[
                local p=A:hadd(B):hscale(1/2)
                    :multiply(objs.view:inverse())
                return objs.curve_visible(p)
            ]]
            return {
                {
                    surface="S",
                    keep="both",
                    intersection_options=S_options,
                    filter=curve_filter
                },
                {
                    surface="O",
                    keep="both",
                    intersection_options=O_options
                }
            }
        }
    ]
    \ltdtdisplaysimplices
\end{tikzpicture}}
\end{document}
