\documentclass[tikz,border=1cm]{standalone}
\usepackage{lua-tikz3dtools}

\begin{document}

\foreach \i in {1,...,24} {
\begin{tikzpicture}
    \useasboundingbox (-4.5,-4.5) rectangle (4.5,4.5);

    \ltdtsetobject[
        name={return "view"},
        object={
            return Matrix.zyzrotation(
                Vector:new{3.75*pi/6+\i*tau/24,pi/3,pi/2}
            )
        }
    ]

    \ltdtsetobject[
        name={return "viewinverse"},
        object={return view:inverse()}
    ]

    \ltdtappendlight[
        v={return Vector:new{1,1,1,1}}
    ]

    % Six hidden radial cylinders used as partitioning surfaces.
    \foreach \k [evaluate=\k as \myangle using \k*pi/3] in {0,...,5} {
        \ltdtappendsurface[
            name=cut\k,
            uparams={return Vector:new{-1.5,1.5,2}},
            vparams={return Vector:new{0,tau,30}},
            v={
                local a = \myangle
                local s = Vector.sphere(Vector:new{a+pi/2,v,0.5})
                return Vector:new{
                    (u+3)*cos(a)+s[1],
                    (u+3)*sin(a)+s[2],
                    s[3],
                    1
                }
            },
            transformation={return view},
            filter={return false}
        ]
    }

    \ltdtappendsurface[
        uparams={return Vector:new{0,tau,20}},
        vparams={return Vector:new{0,tau,14}},
        v={
            local s = Vector.sphere(Vector:new{u,v,1})
            return Vector:new{
                3*cos(u)+s[1],
                3*sin(u)+s[2],
                s[3],
                1
            }
        },
        transformation={return view},
        fill options={
            return function(lighting)
                local percent = math.floor(lighting*100)
                return
                    "preaction={fill opacity=1,fill=white!"
                    .. percent
                    .. "!black},"
                    .. "postaction={draw=none,line width=.2pt,"
                    .. "line join=round,line cap=round}"
            end
        },
        partition by={
            local partitions = {}
            for k=0,5 do
                table.insert(partitions, {
                    surface = "cut" .. k,
                    keep = "both",
                    intersection_options = "draw=red,line width=.6pt"
                })
            end
            return partitions
        },
        filter={
            local p =
                A:hadd(B)
                 :hadd(C)
                 :hscale(1/3)
                 :multiply(viewinverse)
            for k=0,5 do
                local a = k*pi/3
                if p:hdistance(
                    Vector:new{
                        4*cos(a),
                        4*sin(a),
                        0,
                        1
                    }
                ) <= 0.499 then
                    return false
                end
                if p:hdistance(
                    Vector:new{
                        2*cos(a),
                        2*sin(a),
                        0,
                        1
                    }
                ) <= 0.499 then
                    return false
                end
            end
            return true
        }
    ]
    \ltdtdisplaysimplices
\end{tikzpicture}
}

\end{document}
