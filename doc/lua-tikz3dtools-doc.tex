\ltdtsetobject[
    name={return "fixedfour"},
    object={
        return function(x)
            x=max(0,min(1,x))
            local n=floor(x*10000+0.5)
            local whole=floor(n/10000)
            local frac=n-whole*10000
            local tail=tostring(frac)
            if frac<10 then tail="000"..tail elseif frac<100 then tail="00"..tail elseif frac<1000 then tail="0"..tail end
            return tostring(whole).."."..tail
        end
    }
]

\begin{figure}
\centering
\begin{tikzpicture}
\useasboundingbox (-80mm,-45mm) rectangle (80mm,45mm);
\pgfmathsetmacro\SSS{3}
\pgfmathsetmacro{\PHASE}{2*pi*58/120}
\ltdtappendlight[v={return Vector:new{1,1,2,1}}]
\ltdtsetobject[name={return "rot"},object={return \PHASE}]
\ltdtsetobject[
    name={return "view"},
    object={return Matrix.zyzrotation(Vector:new{pi/6,pi/3,pi/2}):multiply(Matrix.perspective(Vector:new{0,0,-1/8}))}
]
\ltdtsetobject[
    name={return "clifford"},
    object={
        return function(u,v)
            local scale=1/sqrt(2)
            local x1=scale*cos(u)
            local y1=scale*sin(u)
            local x2=scale*cos(v)
            local y2=scale*sin(v)
            local xr=x1*cos(rot)-y2*sin(rot)
            local wr=x1*sin(rot)+y2*cos(rot)
            local yr=y1
            local zr=x2
            local denom=1-wr
            if denom<1/40 then return nil end
            local px=xr/denom
            local py=yr/denom
            local pz=zr/denom
            if px*px+py*py+pz*pz>40 then return nil end
            return Vector:new{px,py,pz,1}
        end
    }
]
\ltdtappendsurface[
    uparams={return Vector:new{0,tau,20}},
    vparams={return Vector:new{0,tau,15}},
    transformation={return view},
    v={return clifford(u,v)},
    filter={
        local M=A:hadd(B):hadd(C):hscale(1/3):multiply(view:inverse())
        return abs(M[1])<\SSS+0.001 and abs(M[2])<\SSS+0.001 and abs(M[3])<\SSS+0.001
    },
    fill options={
        local r=0.5*lighting
        local g=0.5+0.5*lighting
        return "fill={rgb,1:red,"..fixedfour(r)..";green,"..fixedfour(g)..";blue,"..fixedfour(g).."},fill opacity=0.9,postaction={draw=black,line cap=round,line join=round}"
    }
]
\ltdtappendsolid[
    uparams={return Vector:new{-\SSS,\SSS,\SSS}},
    vparams={return Vector:new{-\SSS,\SSS,\SSS}},
    wparams={return Vector:new{-\SSS,\SSS,\SSS}},
    transformation={return view},
    filter={return false},
    v={return Vector:new{u,v,w,1}}
]
\ltdtdisplaysimplices
\end{tikzpicture}
\caption{A stereographic image of the Clifford torus.}
\end{figure}
